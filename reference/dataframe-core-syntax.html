groupBy + agg

from pyspark.sql.functions import col, sum, count, avg, max, min

df.groupBy("region").count()                 # region, count
df.groupBy("region").sum("amount")           # region, sum(amount)
df.groupBy("region", "status").count()

df.groupBy("region").agg(
    count("*").alias("n"),
    sum("amount").alias("total"),
    avg("amount").alias("avg_amt"),
    max("amount").alias("hi"),
)

# pre-agg filter (rows) vs post-agg filter (groups / HAVING)
df.filter(col("status") == "open").groupBy("region").agg(...)
df.groupBy("region").agg(sum("amount").alias("total")).filter(col("total") > 280)

Window

from pyspark.sql import Window
from pyspark.sql.functions import col, row_number, rank, dense_rank, lag, lead, sum

w = Window.partitionBy("region").orderBy(col("amount").desc(), col("id"))
df.withColumn("rn", row_number().over(w))
df.withColumn("r", rank().over(Window.partitionBy("region").orderBy(col("amount").desc())))
df.withColumn("d", dense_rank().over(Window.partitionBy("region").orderBy(col("amount").desc())))

w_day = Window.partitionBy("region").orderBy("day")
df.withColumn("prev", lag("amount").over(w_day))
df.withColumn("nxt", lead("amount").over(w_day))
df.withColumn("running", sum("amount").over(w_day))   # default ordered frame

w_all = Window.partitionBy("region")                  # no order → whole partition
df.withColumn("reg_total", sum("amount").over(w_all))

# top-1 per region
df.withColumn("rn", row_number().over(w)).filter(col("rn") == 1)

Window · row_number · lag

groupBy collapses; window keeps rows and adds columns.

Nulls

from pyspark.sql.functions import col, lit, coalesce, when

col("amount").isNull()
col("amount").isNotNull()
# not: col("amount") == None

df.filter(col("status") == "open")          # null status dropped
df.filter(col("status").isNull() | (col("status") == "open"))

df.na.drop(subset=["amount", "status"])     # alias: dropna
df.na.fill({"amount": 0})                   # alias: fillna
df.na.fill(0, subset=["amount"])

df.withColumn("r", coalesce(col("region"), lit("unknown")))
df.select(
    "id",
    when(col("amount").isNull(), "missing").otherwise("present").alias("flag"),
)

col("a").eqNullSafe(col("b"))               # null = null is true
